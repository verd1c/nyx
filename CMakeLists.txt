# Inspect the actual link properties so a new dependency cannot bypass the
# declared foundation boundaries by changing only target_link_libraries.
foreach(target IN ITEMS nyx_support nyx_format nyx_ir nyx_eval nyx_a64 nyx_toy nyx_recovery nyx_analysis nyx_passes nyx_verify nyx)
  set(allowed nyx_options)
  if(target STREQUAL "nyx")
    list(APPEND allowed nyx_format nyx_a64 nyx_ir nyx_eval nyx_recovery nyx_analysis nyx_passes nyx_verify nyx_support)
  elseif(target STREQUAL "nyx_passes")
    list(APPEND allowed nyx_analysis nyx_recovery)
  elseif(target STREQUAL "nyx_eval" OR target STREQUAL "nyx_a64" OR target STREQUAL "nyx_toy" OR target STREQUAL "nyx_recovery" OR target STREQUAL "nyx_analysis")
    list(APPEND allowed nyx_ir)
  elseif(target STREQUAL "nyx_ir")
    list(APPEND allowed nyx_support)
  endif()
  get_target_property(dependencies ${target} LINK_LIBRARIES)
  foreach(dependency IN LISTS dependencies)
    if(NOT dependency IN_LIST allowed)
      message(FATAL_ERROR "Undeclared dependency: ${target} -> ${dependency}")
    endif()
  endforeach()
endforeach()
